cmake_minimum_required(VERSION 3.26.4)
project(BertNativeRuntime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

include("${CMAKE_CURRENT_LIST_DIR}/cmake/BertDependencies.cmake")
option(BERT_BUILD_GPU "Build CUDA runners and prepared engines (OFF needs no CUDA/Poseidon dependencies)" ON)
option(BERT_BUILD_PREPARED "Enable prepared Linear/QK/AV in the single-plan runner" ON)
option(BERT_BUILD_MULTI_PLAN "Also build the same-process multi-plan diagnostic" OFF)
option(BERT_BUILD_CPU_METADATA_CHECKS "Build checks requiring real Poseidon CPU encoding/encryption" OFF)

if(BERT_BUILD_TESTS)
    enable_testing()
endif()
if(BERT_BUILD_GPU)
    bert_default_cuda_architectures()
    enable_language(CUDA)
    bert_setup_dependencies(GPU RUNTIME)
elseif(BERT_BUILD_MULTI_PLAN)
    message(FATAL_ERROR "BERT_BUILD_MULTI_PLAN requires BERT_BUILD_GPU=ON")
endif()

if(BERT_BUILD_PREPARED OR BERT_BUILD_MULTI_PLAN OR NOT BERT_BUILD_GPU)
    # The CPU-only configuration builds the independent arithmetic/layout checks,
    # without finding CUDA, GMP, or any historical Poseidon libraries.
    set(COMPACT_BUILD_GPU "${BERT_BUILD_GPU}" CACHE BOOL "Build prepared Linear CUDA engine")
    set(ATTENTION_BUILD_GPU "${BERT_BUILD_GPU}" CACHE BOOL "Build prepared QK engine")
    set(COMPACT_AV_BUILD_GPU "${BERT_BUILD_GPU}" CACHE BOOL "Build prepared AV engine")
    set(COMPACT_BUILD_SETUP_CHECK "${BERT_BUILD_CPU_METADATA_CHECKS}" CACHE BOOL "Build real CPU Linear checks")
    set(ATTENTION_BUILD_METADATA_CHECK "${BERT_BUILD_CPU_METADATA_CHECKS}" CACHE BOOL "Build real CPU QK check")
    set(COMPACT_AV_BUILD_METADATA_CHECK "${BERT_BUILD_CPU_METADATA_CHECKS}" CACHE BOOL "Build real CPU AV checks")
    if(NOT BERT_BUILD_GPU AND (COMPACT_BUILD_GPU OR ATTENTION_BUILD_GPU OR COMPACT_AV_BUILD_GPU))
        message(FATAL_ERROR "BERT_BUILD_GPU=OFF conflicts with cached per-engine GPU options; use a fresh build directory or disable them")
    endif()
    if(BERT_BUILD_CPU_METADATA_CHECKS AND NOT BERT_BUILD_GPU)
        bert_setup_dependencies()
    endif()
    add_subdirectory(compact_native)
    add_subdirectory(compact_attention)
    add_subdirectory(compact_av)
endif()

if(BERT_BUILD_GPU)
    add_executable(bert_native_runner native_runner.cpp)
    target_link_libraries(bert_native_runner PRIVATE bert::runtime_gpu_api bert::plan_json)
    if(BERT_BUILD_PREPARED)
        if(NOT TARGET bert::compact_linear OR NOT TARGET bert::prepared_qk OR NOT TARGET bert::prepared_av)
            message(FATAL_ERROR "BERT_BUILD_PREPARED requires all three per-engine GPU build options")
        endif()
        target_compile_definitions(bert_native_runner PRIVATE BERT_COMPACT_LINEAR=1 BERT_PREPARED_QK=1 BERT_PREPARED_AV=1)
        target_link_libraries(bert_native_runner PRIVATE bert::compact_linear bert::prepared_qk bert::prepared_av)
    endif()
    set_target_properties(bert_native_runner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    if(BERT_BUILD_MULTI_PLAN)
        add_subdirectory(multi_plan)
    endif()
endif()
