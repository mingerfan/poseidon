cmake_minimum_required(VERSION 3.26.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BertCompactLinear LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/BertDependencies.cmake")
bert_module_paths(COMPACT)
option(COMPACT_BUILD_SETUP_CHECK "Build CPU-only real encoding/compression checks" ON)
option(COMPACT_BUILD_GPU "Build the CUDA engine and opt-in standalone GPU probe" ON)
option(COMPACT_BUILD_PERIODIC_EXPERIMENT "Build CPU-only small-ring/native-encoder oracle (engine stays strict)" OFF)

if(COMPACT_BUILD_GPU)
    bert_default_cuda_architectures()
    enable_language(CUDA)
    bert_setup_dependencies(PREFIX COMPACT GPU)
elseif(COMPACT_BUILD_SETUP_CHECK OR COMPACT_BUILD_PERIODIC_EXPERIMENT)
    bert_setup_dependencies(PREFIX COMPACT)
endif()

if(BERT_BUILD_TESTS)
    enable_testing()
    add_executable(compact_layout_check layout_check.cpp)
    add_test(NAME compact_layout_check COMMAND compact_layout_check)
    if(COMPACT_BUILD_SETUP_CHECK)
        add_executable(compact_setup_check setup_check.cpp)
        target_link_libraries(compact_setup_check PRIVATE bert::cpu)
        add_test(NAME compact_setup_check COMMAND compact_setup_check)
        add_executable(compact_scale_check scale_check.cpp)
        target_link_libraries(compact_scale_check PRIVATE bert::cpu)
        add_test(NAME compact_scale_check COMMAND compact_scale_check)
    endif()
    if(COMPACT_BUILD_PERIODIC_EXPERIMENT)
        add_executable(compact_periodic_encoder_check periodic_encoder_check.cpp)
        target_link_libraries(compact_periodic_encoder_check PRIVATE bert::cpu)
        add_test(NAME compact_periodic_encoder_check COMMAND compact_periodic_encoder_check --synthetic-validation --terms 8)
    endif()
endif()

if(COMPACT_BUILD_GPU)
    add_library(bert_compact_linear STATIC compact_linear.cpp compact_mac.cu)
    add_library(bert::compact_linear ALIAS bert_compact_linear)
    target_include_directories(bert_compact_linear PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
    target_link_libraries(bert_compact_linear PUBLIC bert::gpu)
    # Inherit the user's CMAKE_CUDA_ARCHITECTURES; do not force SM80.
    set_target_properties(bert_compact_linear PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(bert_compact_linear PRIVATE
        $<$<COMPILE_LANGUAGE:CUDA>:--default-stream=per-thread>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
    add_executable(compact_linear_probe probe.cpp)
    target_link_libraries(compact_linear_probe PRIVATE bert::compact_linear)
endif()
