cmake_minimum_required(VERSION 3.26.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BertPreparedAv LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/BertDependencies.cmake")
bert_module_paths(COMPACT_AV)
option(COMPACT_AV_BUILD_METADATA_CHECK "Build genuine CPU encrypted scale/metadata test" ON)
option(COMPACT_AV_BUILD_GPU "Compile/link prepared GPU engine and opt-in probe; never run GPU automatically" ON)
option(COMPACT_AV_BUILD_PACKED_CHECKS "Build explicitly partial PackedInput2 CPU scalar/metadata/mask tests" ON)

if(COMPACT_AV_BUILD_GPU)
    bert_default_cuda_architectures()
    enable_language(CUDA)
    bert_setup_dependencies(PREFIX COMPACT_AV GPU)
elseif(COMPACT_AV_BUILD_METADATA_CHECK)
    bert_setup_dependencies(PREFIX COMPACT_AV)
endif()

if(BERT_BUILD_TESTS)
    enable_testing()
    add_executable(compact_av_math_check math_check.cpp)
    add_test(NAME compact_av_math_check COMMAND compact_av_math_check)
    if(COMPACT_AV_BUILD_PACKED_CHECKS)
        add_executable(compact_av_packed_math_check packed_input_math_check.cpp)
        add_test(NAME compact_av_packed_math_check COMMAND compact_av_packed_math_check)
    endif()
    if(COMPACT_AV_BUILD_METADATA_CHECK)
        add_executable(compact_av_metadata_check metadata_check.cpp)
        target_link_libraries(compact_av_metadata_check PRIVATE bert::cpu)
        add_test(NAME compact_av_metadata_check COMMAND compact_av_metadata_check)
        set_tests_properties(compact_av_metadata_check PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2")
        if(COMPACT_AV_BUILD_PACKED_CHECKS)
            add_executable(compact_av_packed_metadata_check packed_input_metadata_check.cpp)
            target_link_libraries(compact_av_packed_metadata_check PRIVATE bert::cpu)
            add_test(NAME compact_av_packed_metadata_check COMMAND compact_av_packed_metadata_check)
            add_executable(compact_av_packed_masks_check packed_input_masks_check.cpp)
            target_link_libraries(compact_av_packed_masks_check PRIVATE bert::cpu)
            add_test(NAME compact_av_packed_masks_check COMMAND compact_av_packed_masks_check)
            set_tests_properties(compact_av_packed_metadata_check compact_av_packed_masks_check PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2")
        endif()
    endif()
endif()

if(COMPACT_AV_BUILD_GPU)
    add_library(bert_prepared_av STATIC prepared_av.cpp)
    add_library(bert::prepared_av ALIAS bert_prepared_av)
    target_include_directories(bert_prepared_av PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
    target_link_libraries(bert_prepared_av PUBLIC bert::gpu)
    add_executable(compact_av_probe probe.cpp)
    target_link_libraries(compact_av_probe PRIVATE bert::prepared_av)
    if(BERT_BUILD_TESTS)
        add_executable(compact_av_api_check api_check.cpp)
        target_link_libraries(compact_av_api_check PRIVATE bert::prepared_av)
        add_test(NAME compact_av_api_check COMMAND compact_av_api_check)
        add_test(NAME compact_av_probe_help COMMAND compact_av_probe --help)
    endif()
endif()
