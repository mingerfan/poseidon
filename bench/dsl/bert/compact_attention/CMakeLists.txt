cmake_minimum_required(VERSION 3.26.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BertPreparedAttention LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/BertDependencies.cmake")
bert_module_paths(ATTENTION)
option(ATTENTION_BUILD_METADATA_CHECK "Build genuine CPU encrypted scale/metadata test" ON)
option(ATTENTION_BUILD_GPU "Compile/link prepared GPU engine and opt-in probe; never run GPU automatically" ON)

if(ATTENTION_BUILD_GPU)
    bert_default_cuda_architectures()
    enable_language(CUDA)
    bert_setup_dependencies(PREFIX ATTENTION GPU)
elseif(ATTENTION_BUILD_METADATA_CHECK)
    bert_setup_dependencies(PREFIX ATTENTION)
endif()

if(BERT_BUILD_TESTS)
    enable_testing()
    add_executable(attention_math_check math_check.cpp)
    add_test(NAME attention_math_check COMMAND attention_math_check)
    add_executable(attention_key_plan_check key_plan_check.cpp)
    add_test(NAME attention_key_plan_check COMMAND attention_key_plan_check)
    if(ATTENTION_BUILD_METADATA_CHECK)
        add_executable(attention_metadata_check metadata_check.cpp)
        target_link_libraries(attention_metadata_check PRIVATE bert::cpu)
        add_test(NAME attention_metadata_check COMMAND attention_metadata_check)
        set_tests_properties(attention_metadata_check PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2")
    endif()
endif()

if(ATTENTION_BUILD_GPU)
    # The callback uses the C++ compiler; only the optional SOURCE dependencies
    # build core CUDA kernels. No GPU is queried or executed during the build.
    add_library(bert_prepared_qk STATIC prepared_qk.cpp)
    add_library(bert::prepared_qk ALIAS bert_prepared_qk)
    target_include_directories(bert_prepared_qk PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
    target_link_libraries(bert_prepared_qk PUBLIC bert::gpu)
    add_executable(attention_qk_probe probe.cpp)
    target_link_libraries(attention_qk_probe PRIVATE bert::prepared_qk)
    if(BERT_BUILD_TESTS)
        add_test(NAME attention_qk_probe_help COMMAND attention_qk_probe --help)
        add_test(NAME attention_qk_probe_paired_help COMMAND attention_qk_probe --pair-products --help)
        add_test(NAME attention_qk_probe_packed_input_help COMMAND attention_qk_probe --pack-inputs --help)
        add_test(NAME attention_qk_probe_gate COMMAND "${CMAKE_COMMAND}"
            "-DPROBE=$<TARGET_FILE:attention_qk_probe>" -P "${CMAKE_CURRENT_LIST_DIR}/probe_gate_check.cmake")
    endif()
endif()
