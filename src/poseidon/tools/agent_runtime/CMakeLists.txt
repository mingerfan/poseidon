cmake_minimum_required(VERSION 3.22)
project(PoseidonAgentRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(POSEIDON_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT DEFINED POSEIDON_CPU_BUILD)
 message(FATAL_ERROR "Set POSEIDON_CPU_BUILD to the existing native Poseidon build")
endif()
set(CKKS_RUNTIME_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(CKKS_RUNTIME_BUILD_EXPERIMENTS OFF CACHE BOOL "" FORCE)
set(CKKS_RUNTIME_BUILD_MPI OFF CACHE BOOL "" FORCE)
enable_testing()
add_subdirectory("${POSEIDON_ROOT}/third_party/ckks-runtime-agent" runtime)
add_library(poseidon_existing SHARED IMPORTED)
set_target_properties(poseidon_existing PROPERTIES IMPORTED_LOCATION "${POSEIDON_CPU_BUILD}/libposeidon_shared.so"
 INTERFACE_INCLUDE_DIRECTORIES "${POSEIDON_ROOT}/src;${POSEIDON_CPU_BUILD}")
add_library(poseidon_runtime_cpu_api STATIC "${POSEIDON_ROOT}/src/poseidon/runtime_api/poseidon_cpu_api.cpp")
target_link_libraries(poseidon_runtime_cpu_api PUBLIC poseidon_existing ckks_runtime::core)
add_executable(poseidon_agent_cpu cpu_runner.cpp)
target_link_libraries(poseidon_agent_cpu PRIVATE poseidon_runtime_cpu_api ckks_runtime::plan_json)
