cmake_minimum_required(VERSION 3.22)
project(qwen_cpu_parallel LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED COMPONENTS CXX)
set(POSEIDON_BUILD_DEPS OFF CACHE BOOL "" FORCE)
set(POSEIDON_USE_ZLIB OFF CACHE BOOL "" FORCE)
set(POSEIDON_USE_ZSTD OFF CACHE BOOL "" FORCE)
set(POSEIDON_USE_MSGSL OFF CACHE BOOL "" FORCE)
set(POSEIDON_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(POSEIDON_BUILD_CKKS_RUNTIME_API OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/source" "${CMAKE_BINARY_DIR}/poseidon")
target_compile_definitions(poseidon_shared PUBLIC POSEIDON_CPU_RNS_PARALLEL=1)
target_link_libraries(poseidon_shared PUBLIC OpenMP::OpenMP_CXX)
foreach(name bootstrap_parallel_bench qwen_encoder_test qwen_rotation_test)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE poseidon_shared OpenMP::OpenMP_CXX)
  target_include_directories(${name} PRIVATE "${CMAKE_BINARY_DIR}/poseidon" "${CKKS_RUNTIME_SOURCE}/third_party")
endforeach()
enable_testing()
add_test(NAME encoder COMMAND qwen_encoder_test)
add_test(NAME rotation COMMAND qwen_rotation_test)
