def no_running_previous():
    matches=[]
    paths={str(OLD_TOOL),str(Path(__file__).resolve()),
           str(ROOT/"scripts/baseline/run_candidate.py")}
    for line in subprocess.check_output(["/bin/ps","-eo","pid=,args="],text=True,timeout=5).splitlines():
        pid,args=line.strip().split(None,1)
        try: tokens=shlex.split(args)
        except ValueError:continue
        if int(pid)!=os.getpid() and any(x in paths for x in tokens):matches.append(int(pid))
    require(not matches,"Existing authorized execution process")
