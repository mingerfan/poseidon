import copy,json,math,sys,tempfile,unittest
from pathlib import Path
from unittest import mock
import numpy as np
import torch
from benchmark_graph import digest
from model_weight_package import regular_bytes
from pytorch_weight_package import export_state
from hecate_python_env import WORK
from source_precision_reference import run,check_report,SourceReferenceError
from test_model_output_contract import model,tensor
FORWARD_REPORTS=[]

class SourcePrecisionTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory(prefix="source_precision_test_",dir=WORK/"results")
        self.root=Path(self.temp.name)
    def tearDown(self):self.temp.cleanup()
    def fixture(self,forward="return x*x",*,init="pass",state=None,shape=(2,),names=("out",),
                version="poseidon-trusted-pytorch-v12",factory=None,helper=None,tree=None):
        source="import torch\nfrom torch import nn\n"
        if helper:source+="from helper import double\n"
        source+="class Model(nn.Module):\n    def __init__(self, **kwargs):\n        super().__init__()\n        "+init+"\n"
        source+="    def forward(self, x):\n        "+forward.replace("\n","\n        ")+"\n"
        source+="def make(**kwargs): return Model(**kwargs).eval()\n"
        (self.root/"refmodel.py").write_text(source)
        files={"refmodel.py":__import__("hashlib").sha256(source.encode()).hexdigest()}
        if helper:
            (self.root/"helper.py").write_text(helper)
            files["helper.py"]=__import__("hashlib").sha256(helper.encode()).hexdigest()
        weights={} if state is None else export_state(state,self.root/"weight_stage")
        if state is not None:
            for item in weights.values():
                (self.root/item["file"]).write_bytes((self.root/"weight_stage"/item["file"]).read_bytes())
        graph=model(shape=shape,names=names,profile="general-tensors-v1")
        spec=dict(format=version,files=files,entry="refmodel.make",inputs=graph["inputs"],
            outputs=list(names),approximations=dict(format='poseidon-pytorch-approximations-v1',defaults={},modules=[],nodes=[]),factory=factory or dict(args=[],kwargs={}),
            dtype_policy="convert-float64-v1",weights=weights)
        path=self.root/"manifest.json";path.write_text(json.dumps(spec))
        return path,spec,graph,tree or tensor(names[0])
    def evaluate(self,fixture,samples=None,**kwargs):
        path,spec,graph,tree=fixture
        report=run(path,graph,samples or [dict(x=[-.13,.3])],self.root/"evaluation",
            expected_manifest_sha256=digest(spec),input_dtypes={"x":"float64"},output_structure=tree,
            trusted=True,**kwargs)
        FORWARD_REPORTS.append(report["binding"]);return report
    def rejected(self,code,fn):
        with self.assertRaises(SourceReferenceError) as caught:fn()
        self.assertEqual(caught.exception.code,code)
        self.assertNotIn("private",str(caught.exception))
    def test_float32_linear_preserves_source_weight_and_input_precision(self):
        state=torch.nn.Module();state.layer=torch.nn.Linear(2,2,dtype=torch.float32)
        with torch.no_grad():
            state.layer.weight.copy_(torch.tensor([[-.375,.625],[.22,.43]],dtype=torch.float32))
            state.layer.bias.copy_(torch.tensor([.17,-.29],dtype=torch.float32))
        f=self.fixture("return self.layer(x)",init="self.layer=nn.Linear(2,2,dtype=torch.float32)",state=state)
        path,spec,graph,tree=f
        samples=[dict(x=[-.13,.3]),dict(x=[.31,-.57]),dict(x=[0.,0.])]
        before=copy.deepcopy(samples)
        report=run(path,graph,samples,self.root/"evaluation",expected_manifest_sha256=digest(spec),
            input_dtypes={"x":"float32"},output_structure=tree,trusted=True)
        FORWARD_REPORTS.append(report["binding"])
        for sample,actual in zip(samples,report["outputs"]):
            expected=state.layer(torch.tensor(sample["x"],dtype=torch.float32)).detach().numpy()
            np.testing.assert_array_equal(actual["out"],expected.astype(np.float64))
        self.assertEqual(report["output_dtypes"],{"out":"float32"})
        self.assertEqual(set(report["source_state_dtypes"].values()),{"float32"})
        self.assertGreater(report["max_input_conversion_absolute_error"]["x"],0.)
        self.assertEqual(samples,before);self.assertTrue(report["native_dtype_preserved"])
        self.assertFalse(report["graph_semantic_equivalence_certified"])
    def test_exact_float64_and_no_input_rounding(self):
        r=self.evaluate(self.fixture())
        np.testing.assert_array_equal(r["outputs"][0]["out"],np.square(np.array([-.13,.3])))
        self.assertEqual(r["max_input_conversion_absolute_error"],{"x":0.})
        self.assertEqual(r["output_dtypes"],{"out":"float64"})
    def test_gelu_tanh_executes_source_variant(self):
        r=self.evaluate(self.fixture("return torch.nn.functional.gelu(x, approximate='tanh')"),
                        [dict(x=[-1.3,.44])])
        x=torch.tensor([-1.3,.44],dtype=torch.float64)
        expected=torch.nn.functional.gelu(x,approximate="tanh").numpy()
        np.testing.assert_array_equal(r["outputs"][0]["out"],expected)
        erf=np.array([v*.5*(1+math.erf(v/math.sqrt(2))) for v in x.numpy()])
        self.assertGreater(np.max(np.abs(expected-erf)),1e-6)
    def test_nested_dictionary_tuple_list_and_none(self):
        tree=dict(kind="dict",items=[dict(key="prediction",value=tensor("a")),
            dict(key="features",value=dict(kind="tuple",items=[tensor("b"),
                dict(kind="list",items=[tensor("c"),dict(kind="none")])]))])
        f=self.fixture("return {'prediction':x*x, 'features':(x+x,[x-.25,None])}",
                       names=("a","b","c"),version="poseidon-trusted-pytorch-v16",tree=tree)
        r=self.evaluate(f)
        self.assertEqual(r["output_structure"],tree)
        np.testing.assert_array_equal(r["outputs"][0]["c"],np.array([-.13,.3])-.25)
    def test_scalar_output_abi(self):
        tree=dict(kind="tuple",items=[tensor("out"),tensor("sum",scalar=True)])
        f=self.fixture("return x*x,x.sum()",shape=(1,),names=("out","sum"),tree=tree)
        r=self.evaluate(f,[dict(x=[.3])]);self.assertEqual(r["outputs"][0]["sum"],[.3])
    def test_factory_parameters_and_multifile_helper(self):
        f=self.fixture("return double(x)+self.offset",init="self.offset=kwargs['offset']",
            factory=dict(args=[],kwargs={"offset":.375}),helper="def double(x): return x+x\n")
        r=self.evaluate(f);np.testing.assert_array_equal(r["outputs"][0]["out"],np.array([-.13,.3])*2+.375)
    def test_registered_integer_and_boolean_buffers_preserve_dtype(self):
        state=torch.nn.Module()
        state.register_buffer("indices",torch.tensor([1,0],dtype=torch.int64))
        state.register_buffer("mask",torch.tensor([True,False],dtype=torch.bool))
        f=self.fixture("return x.index_select(0,self.indices)+self.mask.to(x.dtype)",
            init="self.register_buffer('indices',torch.tensor([1,0],dtype=torch.int64));self.register_buffer('mask',torch.tensor([True,False]))",
            state=state)
        r=self.evaluate(f)
        self.assertEqual(r["source_state_dtypes"],{"indices":"int64","mask":"bool"})
        np.testing.assert_array_equal(r["outputs"][0]["out"],[1.3,-.13])
    def test_input_alias_mutation_uses_owned_storage(self):
        samples=[dict(x=[.1,-.2])];saved=copy.deepcopy(samples)
        r=self.evaluate(self.fixture("x.mul_(2)\nreturn x*x"),samples)
        self.assertEqual(samples,saved);self.assertEqual(r["input_mutation_observed"],{"x":True})
        np.testing.assert_array_equal(r["outputs"][0]["out"],np.square(np.array([.1,-.2])*2))
    def test_random_forward_is_rejected(self):
        self.rejected("source_reference_random_forward",
                      lambda:self.evaluate(self.fixture("return x+torch.rand_like(x)")))
    def test_public_state_mutation_is_rejected(self):
        state=torch.nn.Module();state.register_buffer("offset",torch.zeros(2,dtype=torch.float64))
        f=self.fixture("self.offset.add_(1)\nreturn x+self.offset",
            init="self.register_buffer('offset',torch.zeros(2,dtype=torch.float64))",state=state)
        self.rejected("source_reference_state_mutation",lambda:self.evaluate(f))
    def test_source_dtype_mismatch_is_explicit_failure(self):
        state=torch.nn.Module();state.layer=torch.nn.Linear(2,2,dtype=torch.float32)
        f=self.fixture("return self.layer(x)",init="self.layer=nn.Linear(2,2,dtype=torch.float32)",state=state)
        self.rejected("source_reference_forward_rejected",lambda:self.evaluate(f))
    def test_host_marker_cannot_be_read_in_sandbox(self):
        marker=self.root/"host-marker";marker.write_text("not a credential")
        f=self.fixture("open("+repr(str(marker))+",'r').read()\nreturn x*x")
        self.rejected("source_reference_forward_rejected",lambda:self.evaluate(f))
        self.assertEqual(marker.read_text(),"not a credential")
    def test_output_structure_mismatch_is_rejected(self):
        f=self.fixture("return {'different':x*x}")
        self.rejected("source_reference_output_structure",lambda:self.evaluate(f))
    def test_no_model_import_or_worker_without_trust(self):
        path,spec,graph,tree=self.fixture("raise RuntimeError('must not execute')")
        with mock.patch("source_precision_reference.subprocess.Popen") as process:
            self.rejected("source_reference_input_rejected",
                lambda:run(path,graph,[dict(x=[0.,0.])],self.root/"evaluation",
                    expected_manifest_sha256=digest(spec),input_dtypes={"x":"float64"},output_structure=tree))
            process.assert_not_called()
        self.assertNotIn("refmodel",sys.modules)
    def test_changed_source_weight_or_manifest_rejected_before_worker(self):
        for kind in ("source","manifest","weight"):
            with self.subTest(kind=kind):
                temp=Path(tempfile.mkdtemp(dir=self.root))
                original_root=self.root;self.root=temp
                try:
                    state=torch.nn.Module();state.layer=torch.nn.Linear(2,2,dtype=torch.float32)
                    path,spec,graph,tree=self.fixture("return self.layer(x)",
                        init="self.layer=nn.Linear(2,2,dtype=torch.float32)",state=state)
                    if kind=="source":(self.root/"refmodel.py").write_text("raise RuntimeError('changed')")
                    if kind=="manifest":
                        changed=copy.deepcopy(spec);changed["factory"]["kwargs"]["new"]=1;path.write_text(json.dumps(changed))
                    if kind=="weight":
                        w=self.root/next(iter(spec["weights"].values()))["file"]
                        raw=w.read_bytes();w.write_bytes(raw[:-1]+bytes([raw[-1]^1]))
                    with mock.patch("source_precision_reference.subprocess.Popen") as process:
                        with self.assertRaises(SourceReferenceError):
                            run(path,graph,[dict(x=[.1,.2])],self.root/"evaluation",
                                expected_manifest_sha256=digest(spec),input_dtypes={"x":"float32"},output_structure=tree,trusted=True)
                        process.assert_not_called()
                finally:self.root=original_root
    def test_invalid_samples_dtypes_and_output_binding_rejected(self):
        path,spec,graph,tree=self.fixture()
        variants=[([],{"x":"float64"},tree),([dict(x=[0.,0.])]*17,{"x":"float64"},tree),
            ([dict(x=[float("nan"),0.])],{"x":"float64"},tree),
            ([dict(x=[0.])],{"x":"float64"},tree),
            ([dict(x=[0.,0.])],{"x":"float16"},tree),
            ([dict(x=[0.,0.])],{"wrong":"float64"},tree),
            ([dict(x=[0.,0.])],{"x":"float64"},tensor("wrong"))]
        for samples,dtypes,outtree in variants:
            with self.subTest(dtypes=dtypes,tree=outtree),mock.patch("source_precision_reference.subprocess.Popen") as process:
                self.rejected("source_reference_input_rejected",
                    lambda:run(path,graph,samples,self.root/"evaluation",expected_manifest_sha256=digest(spec),
                        input_dtypes=dtypes,output_structure=outtree,trusted=True))
                process.assert_not_called()
    def test_existing_output_directory_is_preserved(self):
        f=self.fixture();(self.root/"evaluation").mkdir();(self.root/"evaluation/keep").write_text("keep")
        self.rejected("source_reference_input_rejected",lambda:self.evaluate(f))
        self.assertEqual((self.root/"evaluation/keep").read_text(),"keep")
    def test_timeout_is_controlled_and_old_limits_unchanged(self):
        import subprocess
        f=self.fixture()
        process=mock.Mock(pid=555555,returncode=-9)
        process.wait.side_effect=[subprocess.TimeoutExpired("sandbox",180),None]
        with mock.patch("source_precision_reference.subprocess.Popen",return_value=process),mock.patch("source_precision_reference.os.killpg") as kill:
            self.rejected("source_reference_timeout",lambda:self.evaluate(f))
            kill.assert_called_once()
        from process_resource_limits import limits
        self.assertIn("150",__import__("inspect").getsource(limits))
    def test_sixteen_samples_and_report_protocol_rejections(self):
        r=self.evaluate(self.fixture(),[dict(x=[.1,.2])]*16)
        request=json.loads((self.root/"evaluation/request.json").read_text())
        self.assertEqual(r["samples"],16);self.assertTrue(r["sandbox"]["network_isolated"])
        self.assertNotEqual(r["sandbox"]["network_namespace"],request["host_net_namespace"])
        for field,value in (("samples",True),("input_dtypes",{"x":"float32"}),("encrypted_execution",True),
                            ("outputs",[{"out":[float("nan"),0.]}]*16),("output_structure",tensor("wrong"))):
            changed=copy.deepcopy(r);changed[field]=value
            changed["binding"]=digest({k:v for k,v in changed.items() if k!="binding"})
            with self.subTest(field=field),self.assertRaises((SourceReferenceError,ValueError)):
                check_report(changed,request)
