cmake_minimum_required(VERSION 3.22)
project(PoseidonLLVMConsumerProbe LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS compile=2 link=1)
set(CMAKE_JOB_POOL_COMPILE compile)
set(CMAKE_JOB_POOL_LINK link)
find_package(LLVM 18.1.2 EXACT REQUIRED CONFIG)
add_executable(poseidon_llvm_link_probe probe.cpp)
target_compile_features(poseidon_llvm_link_probe PRIVATE cxx_std_17)
target_include_directories(poseidon_llvm_link_probe SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
llvm_map_components_to_libnames(probe_libs Support)
target_link_libraries(poseidon_llvm_link_probe PRIVATE ${probe_libs})
if(NOT EXISTS "${DACAPO_INCLUDE_DIR}/hecate/Support/HEVMHeader.h")
  message(FATAL_ERROR "The pinned Dacapo header is required for the HEVM ABI probe")
endif()
add_executable(poseidon_hevm_abi_probe hevm_abi.cpp)
target_compile_features(poseidon_hevm_abi_probe PRIVATE cxx_std_17)
target_include_directories(poseidon_hevm_abi_probe PRIVATE "${DACAPO_INCLUDE_DIR}")
