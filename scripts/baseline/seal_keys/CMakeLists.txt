cmake_minimum_required(VERSION 3.22)
project(seal_golden_keys LANGUAGES CXX)
find_package(SEAL 4.0.0 EXACT REQUIRED CONFIG)
add_executable(seal_golden_keys main.cpp)
target_compile_features(seal_golden_keys PRIVATE cxx_std_17)
target_link_libraries(seal_golden_keys PRIVATE SEAL::seal)
add_library(seal_golden_metadata SHARED metadata.cpp)
target_compile_features(seal_golden_metadata PRIVATE cxx_std_17)
target_link_libraries(seal_golden_metadata PRIVATE SEAL::seal)
add_executable(seal_rotation_precision_probe rotation_precision_probe.cpp)
target_compile_features(seal_rotation_precision_probe PRIVATE cxx_std_17)
target_link_libraries(seal_rotation_precision_probe PRIVATE SEAL::seal)
add_executable(seal_packed_keys packed_keys.cpp)
target_compile_features(seal_packed_keys PRIVATE cxx_std_17)
target_link_libraries(seal_packed_keys PRIVATE SEAL::seal)
add_library(seal_packed_metadata SHARED packed_metadata.cpp)
target_compile_features(seal_packed_metadata PRIVATE cxx_std_17)
target_link_libraries(seal_packed_metadata PRIVATE SEAL::seal)
